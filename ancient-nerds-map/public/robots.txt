# Guide for AI crawlers and assistants. GPTBot fetched 13,017 URLs here

# No Crawl-delay: it capped Bing/Yandex at ~1 URL/sec, which is far too slow

# Disallow API endpoints from being indexed
Disallow: /api/
Disallow: /data/basemaps/

# Measurement redirects (Discord CTA click counting) are not content
Disallow: /goto/

# BUT: Google's renderer obeys robots.txt for subresources too. The site
# detail pages fetch /api/sites/{id} after hydration; with it blocked, the
# rendered page hung in its loading state and Google filed it as Soft 404
# (5 of 15 sampled detail pages, 2026-08-16). Longest-match wins, so these
# override the /api/ disallow. The JSON answers carry X-Robots-Tag: noindex,
# so the API responses themselves stay out of the index.
Allow: /api/sites/
Allow: /api/news/screenshots/

# Same rule for the sitemap entry pages, which are client-rendered shells and
# fetch their content after mount. Google's renderer hands a robots-blocked
# fetch() a synthetic 499 response, so with these paths blocked it indexed
# /news.html as "HTTP 499 / Retry" (Live-Test 2026-09-15) and the other
# shells as empty lists. Read-only, cheap, noindex via X-Robots-Tag.
Allow: /api/news/feed
Allow: /api/news/stats
Allow: /api/news/filters
Allow: /api/news/articles
Allow: /api/radar/list
Allow: /api/radar/stats
Allow: /api/radar/map
Allow: /api/theo/public
Allow: /api/v1/openapi.json
Allow: /api/v1/facets
# The source registry feeds /search.html and /globe.html; blocked, Google's
# render of both hung on "LOADING". The popup gate that used to ride on it
# is /api/app/interactive now, which stays under the /api/ disallow.
Allow: /api/sources/

# Allow important data files
Allow: /data/labels.json
Allow: /data/sites/
